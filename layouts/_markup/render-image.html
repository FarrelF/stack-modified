-}} {{- $src = printf "%s%s" $ImageURL $resource.RelPermalink -}} {{- if eq .Page.Site.Params.CDN.Images.Permalinks false -}} {{- $src = $resource.RelPermalink -}} {{- end -}} {{- if (default true
