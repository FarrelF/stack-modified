{{- /* Params: Resource: Hugo image resource object (Optional, for processing) Width: Image width (Required) Height: Image height (Required) Resize: Whether to perform resize (Optional, default: false) Attributes: Map of HTML attributes for the tag */ -}} {{- $resource := .Resource -}} {{- $width := .Width -}} {{- $height := .Height -}} {{- $resize := .Resize -}} {{- $attributes := .Attributes | default dict -}} {{- if and $resize $resource (reflect.IsImageResourceProcessable $resource) $width $height -}} {{ partial "helper/picture.html" (dict "Resource" $resource "Mode" "fill" "Width" $width "Height" $height "Scales" (slice 1 2) "MainScale" 1 "SetImgSrc" true "SetImgSrcset" true "Attributes" $attributes ) }} {{- else -}} {{- if $resource -}} {{- $ImageURL := "" -}} {{- if and (site.Params.CDN.Images.URL) (eq hugo.IsProduction true) -}} {{- $ImageURL = site.Params.CDN.Images.URL -}} {{- end -}} {{- $attributes = merge $attributes (dict "src" (printf "%s%s" $ImageURL $resource.RelPermalink)) -}} {{- end -}} {{- end -}}
