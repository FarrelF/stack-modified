attributes for the tag */ -}} {{- $resource := .Resource -}} {{- $width := .Width -}} {{- $height := .Height -}} {{- $resize := .Resize -}} {{- $mode := lower (.Mode | default "fill") -}} {{- $anchor (reflect.IsImageResourceProcessable $resource) -}} {{- if and $targetWidth (gt $targetWidth $resource.Width) -}}{{- $targetWidth = $resource.Width -}}{{- end -}} {{- if and $targetHeight (gt $targetHeight $resource.Height) -}}{{- $targetHeight = $resource.Height -}}{{- end -}} {{- end -}} {{- if ne $mode "fill" -}} {{- $class := printf "%s image-%s" (index $attributes "class" | default "") $mode -}} {{- $attributes = merge $attributes (dict "class" (strings.TrimSpace $class)) -}} {{- end -}} {{- if and $resize $resource (reflect.IsImageResourceProcessable $resource) $targetWidth $targetHeight -}} {{ partial "helper/picture.html" (dict "Resource" $resource "Mode" $mode "Width" $targetWidth "Height" $targetHeight "Anchor" $anchor "Scales" (slice 1 2) "MainScale" 1 "Hint" $hint "SetImgSrc" true "SetImgSrcset" true "Attributes" $attributes ) }} {{- else -}} {{- $ImageURL := "" -}} {{- if and (site.Params.CDN.Images.URL) (eq hugo.IsProduction true) -}} {{- $ImageURL = site.Params.CDN.Images.URL -}} {{- end -}} {{- $emit := false -}} {{- if $resource -}} {{- $attributes = merge $attributes (dict "src" (printf "%s%s" $ImageURL $resource.RelPermalink)) -}} {{- $emit = true -}} {{- else -}} {{- $emit = true -}} {{- end -}} {{- if $emit -}} {{- end -}} {{- end -}}
