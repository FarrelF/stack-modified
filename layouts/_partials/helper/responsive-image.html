{{- /* Params: Resource: Hugo image resource object (Optional, for processing) Widths: Slice of widths to generate for srcset Attributes: Map of HTML attributes for the tag */ -}} {{- $resource := .Resource -}} {{- $widths := .Widths -}} {{- $attributes := .Attributes | default dict -}} {{- $ImageURL := "" -}} {{- if and site.Params.CDN.Images.URL (eq hugo.IsProduction true) -}} {{- $ImageURL = site.Params.CDN.Images.URL -}} {{- end -}} {{/* Generate srcset if Resource and Widths are provided. Do not append the unprocessed original. */}} {{- if and $widths $resource (reflect.IsImageResourceProcessable $resource) -}} {{- $srcset := slice -}} {{- range $widths -}} {{- if lt . $resource.Width -}} {{- $resized := $resource.Resize (printf "%dx" .) -}} {{- $srcset = $srcset | append (printf "%s%s %dw" $ImageURL $resized.RelPermalink .) -}} {{- end -}} {{- end -}} {{- if gt (len $srcset) 0 -}} {{- $attributes = merge $attributes (dict "srcset" (delimit $srcset ", ")) -}} {{- end -}} {{- end -}}
